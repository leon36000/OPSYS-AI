def lancer(tests, timeout=900):
    r = subprocess.run([str(S/"run-athena.sh"), *tests, "-x", "-q"], capture_output=True, text=True, timeout=timeout)
    return r.returncode, (r.stdout + r.stderr)[-600:]
resultats = []
cible = sys.argv[1:] 
for nom, fichier, ancien, nouveau in MUTATIONS:
    if cible and not any(nom.startswith(c) for c in cible): continue
    sauve = fichier.read_text()
    if sauve.count(ancien) != 1:
        resultats.append((nom, "MOTIF-INTROUVABLE", sauve.count(ancien))); print(resultats[-1], flush=True); continue
    fichier.write_text(sauve.replace(ancien, nouveau))
    try:
        t0 = time.time()
        code, sortie = lancer(UNIT)
        etat = "TUE(unit)" if code != 0 else None
        if etat is None:
            code, sortie = lancer(INTEG)
            etat = "TUE(integ)" if code != 0 else "SURVIT"
        # M8/M10 : tests propres aux départements
        if etat == "SURVIT" and "m08-analytics" in str(fichier):
            code, sortie = lancer(["departments/m08-analytics/tests/test_synthese_direction.py"]); etat = "TUE(m8)" if code != 0 else "SURVIT"
        if etat == "SURVIT" and "m10-memoire" in str(fichier):
            code, sortie = lancer(["departments/m10-memoire/tests/test_strategique.py"]); etat = "TUE(m10)" if code != 0 else "SURVIT"
        if etat == "SURVIT" and "m05-financecore" in str(fichier):
            code, sortie = lancer(["tests/test_m9_athena_raccord.py", "-k", "sources_fiscales"]); etat = "TUE(m5)" if code != 0 else "SURVIT"
        resultats.append((nom, etat, round(time.time()-t0, 1)))
    finally:
        fichier.write_text(sauve)
    print(resultats[-1], flush=True)
